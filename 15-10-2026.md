## q1
```Python
def selection_sort(l): # TC: O(n^2), SC: O(1)
  n = len(l) # len() is TC: O(1)
  if n <= 1:
    return l

  for i in range(n-1):
    min = i
    for j in range(i+1, n):
      if l[j] < l[min]:
        min = j
    # swapping
    l[i], l[min] = l[min], l[i]

  return l
# selection sort is not stable
# selection sort is in-place
```

## q2
```Python
def insertion_sort(l): # TC: O(n^2) (worst, average)
  # TC: O(n) (best case, when a sorted list is passed)
  # SC: O(1)
  n = len(l)
  for i in range(1, n):
    key = l[i]
    j = i-1
    while j >= 0 and l[j] > key:
      l[j+1] = l[j]
      j -= 1
    l[j+1] = key
  return l
# insertion sort is stable
# (but careful about how the key is compared to previous elements)
# insertion sort is in-place
```

# top-down (divide & conquer approach) (recursive)
def merge(left, right): # TC: O(n), SC: O(n)
  arr = []
  l = r = 0
  while l < len(left) and r < len(right):
    if left[l] <= right[r]:
      arr.append(left[l])
      l += 1
    else:
      arr.append(right[r])
      r += 1

  while l < len(left):
    arr.append(left[l])
    l += 1
  # OR
  # arr.extend(left[l:])

  while r < len(right):
    arr.append(right[r])
    r += 1
  # OR
  # arr.extend(right[r:])
  return arr

def merge_sort(l): # TC: O(nlogn), SC: O(n)
  if len(l) <= 1:
    return l
  mid = len(l)//2
  left = merge_sort(l[:mid])
  right = merge_sort(l[mid:])
  return merge(left, right)

# merge sort is stable always
# the above merge sort is not in-place

"""
R(n) = 2*R(n/2) + O(n) -> TC: O(nlogn)
"""

def inplace_merge_sort(l, left, right): # TC: , SC: O(1)
  if left < right:
    mid = left + (right - left) // 2
    inplace_merge_sort(l, left, mid)
    inplace_merge_sort(l, mid+1, right)
    inplace_merge(l, left, mid, right)

def inplace_merge(l, left, mid, right):
  start = mid+1
  if l[mid] <= l[start]:
    return
  
  while left <= mid and start <= right:
    if l[left] <= l[start]:
      left += 1
    else:
      value = l[start]
      index = start
      while index != left:
        l[index] = l[index-1]
        index -= 1
      l[left] = value

      left += 1
      mid += 1
      start += 1

l = [3,4,5,1,2,9,10]
inplace_merge_sort(l, 0, len(l)-1)
