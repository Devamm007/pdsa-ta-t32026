# Unweighted Graph: Adjacency List representation
adjl = {
    '0': ['1', '2'],
    '1': ['0', '3'],
    '2': ['0', '4'],
    '3': ['1'],
    '4': ['2']
}   

# Recursive DFS using Adjacency List
# TC: O(V + E) where V is the number of vertices and E is the number of edges
# SC: O(V) for the visited set + recursion call stack
def DFS(adjl, s, visited=None):
  if visited is None:
    visited = set()
  visited.add(s)
  print(f"{s}")
  for neigh in adjl[s]:
    if neigh not in visited:
      DFS(adjl, neigh, visited)
  return visited

# Test DFS starting from node '2'
DFS(adjl, '2')
# Traversal Output: 2 -> 0 -> 1 -> 3 -> 4
```

from collections import deque

# Unweighted Graph: Adjacency List representation
adjl = {
    '0': ['1', '2'],
    '1': ['0', '3'],
    '2': ['0', '4'],
    '3': ['1'],
    '4': ['2']
}

# --- Approach 1: BFS using a standard Python list as queue ---
# Note: queue.pop(0) takes O(V) time to shift elements, making overall TC: O(V^2 + E)
def BFS_list(adjl, s):
  visited = set()
  queue = [s]
  while queue:
    node = queue.pop(0)
    print(node)
    visited.add(node)
    for neigh in adjl[node]:
      if neigh not in visited:
        queue.append(neigh)
  return visited

# --- Approach 2: BFS using collections.deque (Recommended & Optimal) ---
# TC: O(V + E) because deque.popleft() is O(1) amortized
# SC: O(V) for queue and visited set
def BFS(adjl, s):
  visited = set()
  queue = deque([s])
  while queue:
    node = queue.popleft()
    print(node)
    visited.add(node)
    for neigh in adjl[node]:
      if neigh not in visited:
        queue.append(neigh)
  return visited

# Test BFS starting from node '2'
BFS(adjl, '2')
# Traversal Output: 2 -> 0 -> 4 -> 1 -> 3

"""
========================================================================
NOTE: Why BFS CANNOT Have a Pure Recursive Implementation
========================================================================
1. Recursion inherently uses the Call Stack, which is a LIFO 
   (Last-In, First-Out) data structure.
2. DFS naturally maps to recursion because it goes as deep as possible 
   along a branch before backtracking (LIFO order).
3. BFS, by contrast, explores level-by-level across a frontier, which 
   strictly requires a FIFO (First-In, First-Out) Queue.
4. Any attempt to write "recursive BFS" requires explicitly passing 
   a FIFO queue data structure down through each recursive call. That 
   merely wraps an iterative loop in recursion without using the call 
   stack for exploration. Therefore, BFS cannot have a natural/pure 
   recursive implementation.
========================================================================
"""

from collections import deque

"""
========================================================================
Key Session Takeaway: Connected Components & Reachability
========================================================================
BFS and DFS are fundamental building blocks for graph reachability:
1. Reachability: Can we travel from source 's' to destination 'd'?
   - Run DFS or BFS starting at 's'. If 'd' is in visited, it is reachable.
2. Connected Components: In an undirected graph with multiple disconnected
   subgraphs (islands), a single DFS/BFS only visits one component.
   - By iterating through all vertices and triggering DFS/BFS on any 
     vertex not yet visited, we discover and count all connected components.
========================================================================
"""

# Example: Disconnected Graph with 2 components: {0-1-3} and {2-4}
disconnected_adjl = {
    '0': ['1'],
    '1': ['0', '3'],
    '2': ['4'],
    '3': ['1'],
    '4': ['2']
}

def find_connected_components(adjl): # TC: O(V + E), SC: O(V)
  visited = set()
  components = []

  def dfs_collect(node, comp):
    visited.add(node)
    comp.append(node)
    for neigh in adjl.get(node, []):
      if neigh not in visited:
        dfs_collect(neigh, comp)

  for vertex in adjl:
    if vertex not in visited:
      comp = []
      dfs_collect(vertex, comp)
      components.append(comp)

  return components

def is_reachable(adjl, src, dest): # TC: O(V + E), SC: O(V)
  visited = set()
  queue = deque([src])
  visited.add(src)

  while queue:
    node = queue.popleft()
    if node == dest:
      return True
    for neigh in adjl.get(node, []):
      if neigh not in visited:
        visited.add(neigh)
        queue.append(neigh)
  return False

print("Connected Components:", find_connected_components(disconnected_adjl))
# Output: [['0', '1', '3'], ['2', '4']] (2 components)

print("Is '0' reachable from '4'?:", is_reachable(disconnected_adjl, '4', '0'))
# Output: False
print("Is '0' reachable from '3'?:", is_reachable(disconnected_adjl, '3', '0'))
# Output: True
```
